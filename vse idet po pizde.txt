local CoreGui = game:GetService("CoreGui")
local StarterGui = game:GetService("StarterGui")
local Players = game:GetService("Players")
local LP = Players.LocalPlayer

local discordLink = "https://discord.gg/xpXWpv5Zf"

pcall(function()
    if type(setclipboard) == "function" then
        setclipboard(discordLink)
    elseif type(toclipboard) == "function" then
        toclipboard(discordLink)
    elseif type(syn) == "table" and type(syn.write_clipboard) == "function" then
        syn.write_clipboard(discordLink)
    elseif type(Clipboard) == "table" and type(Clipboard.set) == "function" then
        Clipboard.set(discordLink)
    end
end)

pcall(function()
    StarterGui:SetCore("SendNotification", {
        Title = "PROJECT ASTRA",
        Text = "Discord link copied to clipboard!",
        Duration = 5
    })
end)

pcall(function()
    local old = CoreGui:FindFirstChild("ProjectAstra_Teaser")
    if old then old:Destroy() end
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then
        local old2 = pg:FindFirstChild("ProjectAstra_Teaser")
        if old2 then old2:Destroy() end
    end
end)
